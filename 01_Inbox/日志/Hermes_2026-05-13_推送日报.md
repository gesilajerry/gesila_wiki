## 执行模式（在 execute_code 或 terminal 中）

推荐在 `execute_code` 中用 Python subprocess 调用：

```python
import subprocess, json

def mmx_search(query, max_results=10):
    cmd = f'mmx search query --q "{query}" --output json'
    r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
    data = json.loads(r.stdout)
    results = []
    for item in data.get('organic', [])[:max_results]:
        results.append({
            'title': item.get('title', ''),
            'link': item.get('link', ''),
            'snippet': item.get('snippet', '').replace('</p><p>', ' ').replace('<p>', '').replace('</p>', ''),
            'date': item.get('date', '')
        })
    return results
```

## 搜索多条（多关键词合并去重）

```python
def search_briefing(keywords, max_results=5, max_total=12):
    all_results = {}
    for kw in keywords:
        try:
            results = mmx_search(kw, max_results=max_results)  # 注意参数名是 max_results，不是 max_per_kw
            for r in results:
                # 用 title 做去重 key
                key = r['title'][:30]
                if key not in all_results:
                    all_results[key] = r
        except Exception as e:
            print(f"搜索 '{kw}' 失败: {e}")
    
    items = list(all_results.values())
    items.sort(key=lambda x: x.get('date', ''), reverse=True)
    return items[:max_total]
```

## 飞书 Post 消息发送（踩坑总结）

飞书 post 消息 content 格式极易报 230001 错误（invalid message content），正确格式：

```python
import requests, json

# 正确的 content 格式：数组的数组 [[{...}], [{...}], ...]
# 每个内层数组 = 一个段落（paragraph），外层数组包含所有段落
content = [
    [{"tag": "text", "text": "第一行内容"}],
    [{"tag": "text", "text": "第二行内容"}],
    [{"tag": "text", "text": "第三行内容"}],
]

post_content = {
    "zh_cn": {
        "title": "标题",
        "content": content  # 直接放数组对象，不要再 json.dumps
    }
}

payload = {
    "receive_id": chat_id,
    "msg_type": "post",
    "content": json.dumps(post_content, ensure_ascii=False)
}

r = requests.post(
    "https://open.feishu.cn/open-apis/im/v1/messages?receive_id_type=chat_id",
    headers={
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json; charset=utf-8"
    },
    data=json.dumps(payload, ensure_ascii=False).encode("utf-8"),
    verify=False
)
```

关键点：
- content 是 `[[{...}]]`（数组的数组），不是 `[{...}]`（扁平数组）——错误格式会导致 230001
- content 字段本身是对象（直接放数组），不是字符串；外层 payload.content 才是 json.dumps 后的字符串
- 报错 230001 通常是 content 格式不对，不是 JSON 字符串或 token 问题
- 先用 text 类型消息测试 chat_id 和 token 是否正确
- 用 `requests` 而非 `urllib.request`——Python 3.9 urllib 对飞书文件上传和部分 POST 会持续返回 500，改用 requests 解决

## 新闻监控场景：提高搜索新鲜度的技巧

当搜索近期新闻时，以下策略可显著提升结果新鲜度：

1. **中文日期约束**：在 query 中加入具体日期的汉字形式，如 `"2026年5月9日 纳米"`，比 `"nanotech May 9 2026"` 返回结果更新
2. **中文实体名优先**：搜索中国科技新闻时，用中文专有名词直接搜索（如 `"汉原2号"`、`"本源悟空"`），英文名往往无结果
3. **多轮渐进式搜索**：先搜索宽泛关键词（如 `原子级制造 2026年5月`），再根据发现的实体名做二次精准搜索
4. **snippet 清洗**：返回的 snippet 含 HTML 标签，用 `re.sub(r'<[^>]+>', '', text)` 彻底去除，不要只用简单的字符串 replace
5. **搜索结果为空时的处理**：尝试去掉具体日期，改用 `"2026年5月 纳米制造 突破"` 等月级别约束，避免日期过细导致漏筛

import re, subprocess, json

def mmx_search_news(query, max_results=8):
    """专门用于搜索近期新闻的 mmx search 封装"""

    r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
    try:
        data = json.loads(r.stdout)
        results = []
        for item in data.get('organic', [])[:max_results]:

                'snippet': snippet[:200],
                'date': item.get('date', '')
            })
        return results

        return [{'error': str(e)}]

## 发布流程（批量版）

### 关键注意事项

> **踩坑记录（2026-04/05 实测）：**
> 1. **每次请求必须单独获取 CSRF Token**——Token 不可以跨请求复用，否则 500。Bearer Token 有时效，发布失败先用 GET /api/user 验证 token 是否有效。
> 2. **Bearer Token 有效但 API 返回 401**——此时 CSRF 也必定失效，需重新获取 CSRF 后再重试（不是重新登录）。
> 3. **禁止用 emoji 正则过滤内容**——Python 3.14 的 emoji 范围 `\\U00002600-\\U000026FF` 会误删 CJK 中文字符，导致 HTML 乱码。中文内容原样传即可，无需过滤。
> 4. **标题中的 `×`(U+00D7) 必须替换为 `x`**——服务端 JSON 解析遇到 `×` 会返回 500。
> 5. **500 错误有随机性**，建议加 3 次重试（间隔 3-5 秒）。

### Step 1: 发布函数（含重试 + 单独 CSRF + 可选 forum_id）

```python
import urllib.request, json, ssl, time
from pathlib import Path

ctx = ssl.create_default_context()
ctx.check_hostname = False
ctx.verify_mode = ssl.CERT_NONE
TOKEN_FILE = Path('~/.hermes/skills/blog-publisher/token.txt').expanduser()

def get_token():
    """每次从文件读取最新 Token（支持运行时刷新）"""
    return TOKEN_FILE.read_text().strip()

def publish_one(title, content_html, group_ids, forum_id=2, attempt=1):
    """发布单篇文章，每次单独获取 CSRF。
    forum_id: 板块ID（默认2=综合管理，1=市场销售，4=研发工程）
    """
    TOKEN = get_token()  # ← 每次运行时读取，不硬编码
    try:
        # 每次单独获取 CSRF，不要复用
        csrf_req = urllib.request.Request(
            'https://192.168.30.198:8443/api/csrf',
            headers={'Authorization': f'Bearer {TOKEN}'})
        csrf = json.loads(
            urllib.request.urlopen(csrf_req, context=ctx, timeout=10).read()
        )['data']

        payload = json.dumps({
            "forum_id": forum_id,  # ← 板块ID，不能写死2
            "title": title,
            "content": content_html,
            "group_ids": group_ids,
            "type": "blog"
        }, ensure_ascii=False).encode('utf-8')

        req = urllib.request.Request(
            'https://192.168.30.198:8443/api/discussion',
            data=payload,
            headers={
                'Authorization': f'Bearer {TOKEN}',
                'X-CSRF-Token': csrf,
                'Content-Type': 'application/json; charset=utf-8',
                'Accept': 'application/json'
            },
            method='POST')
        return json.loads(urllib.request.urlopen(req, context=ctx, timeout=15).read())

    except urllib.error.HTTPError as e:
        if e.code == 500 and attempt < 3:
            time.sleep(3)
            return publish_one(title, content_html, group_ids, forum_id, attempt + 1)
        return {'success': False, 'error': f'HTTP {e.code}: {e.reason}'}
```

### Step 2: Markdown 转 HTML

```python
import re

def md_to_html(md_text):
    """不含 emoji 过滤（Python 3.14 regex bug 会损坏 CJK 字符）"""
    html = md_text
    html = re.sub(r'^---\n[\s\S]+?\n---\n', '', html)
    html = re.sub(r'^# (.+)$', '<h1>\\1</h1>', html, flags=re.MULTILINE)
    html = re.sub(r'^## (.+)$', '<h2>\\1</h2>', html, flags=re.MULTILINE)
    html = re.sub(r'^### (.+)$', '<h3>\\1</h3>', html, flags=re.MULTILINE)
    html = re.sub(r'^#### (.+)$', '<h4>\\1</h4>', html, flags=re.MULTILINE)
    html = re.sub(r'^---$', '<hr>', html, flags=re.MULTILINE)
    html = re.sub(r'```(?:\w*)\n([\s\S]*?)\n```', '<pre><code>\\1</code></pre>', html)
    html = re.sub(r'`([^`]+)`', '<code>\\1</code>', html)
    html = re.sub(r'\*\*(.+?)\*\*', '<strong>\\1</strong>', html)
    html = re.sub(r'\*(.+?)\*', '<em>\\1</em>', html)
    html = re.sub(r'^> (.+)$', '<p><em>\\1</em></p>', html, flags=re.MULTILINE)
    # × 符号可能导致 500，标题级别在调用处处理
    lines = html.split('\n')
    in_table = False
    new_lines = []
    for line in lines:
        if '|' in line and line.strip().startswith('|'):
            cells = [c.strip() for c in line.strip().strip('|').split('|')]
            if not in_table:
                new_lines.append('<table border="1" cellpadding="8" cellspacing="0">')
                in_table = True
            is_header = all(re.match(r'[-:]+$', c.replace(' ', '')) for c in cells)
            tag = 'th' if is_header else 'td'
            new_lines.append('<tr>' + ''.join(
                f'<{tag}>{c}</{tag}>' for c in cells
                if not re.match(r'^[-:]+$', c.replace(' ', ''))
            ) + '</tr>')
        else:
            if in_table:
                new_lines.append('</table>')
                in_table = False
            new_lines.append(line)
    if in_table:
        new_lines.append('</table>')
    return '\n'.join(new_lines)

def extract_title(md):
    """提取标题并处理特殊字符"""
    m = re.search(r'^#\s+(.+)$', md, re.MULTILINE)
    title = m.group(1).strip() if m else '无标题'
    title = title.replace('×', 'x')   # × 会导致服务端 500
    title = title.replace('$', '美元')  # $ 也会导致服务端 500
    return title
```

## 常见问题

| 问题 | 原因 | 解决 |
|------|------|------|
| 评论报404 | 接口路径写错 | 必须用 `/api/discussion/{uuid}/comment`，不是 `/api/comment` |
| 404 err:record not found | 用了数字id而非uuid | 评论接口只接受UUID字符串 |
| csrf token mismatch | CSRF Token 为空字符串 | 从响应 `data` 字段提取实际值 |
| 发布成功但列表查不到 | 可能是 group_ids 权限问题 | 确认账号有该分类权限 |
| koala_session httpOnly | 浏览器 cookie 是 httpOnly | 使用 Bearer Token 而非 cookie |
| HTTP 500 随机出现 | CSRF Token 复用或服务端波动 | 每次单独获取 CSRF + 3次重试 |
| HTTP 500 稳定复现（标题含×） | `×`(U+00D7) 导致 JSON 解析失败 | 将 `×` 替换为 `x` 后重试 |
| HTML 变乱码（CJK被删） | Python 3.14 emoji 正则 bug 误删 CJK | 不要用 emoji 过滤函数，原文直传 |
| HTTP 500 err:ratelimit | 请求过于频繁，触发服务端限速 | 批量发布时每篇间隔 ≥10 秒，且必须单篇顺序发，不能并发 |
| HTTP 500 标题含特殊字符 | 标题含 `$`、`×` 等字符导致 JSON 解析失败 | 发布前替换：`.replace('$', '美元').replace('×', 'x')` |

- `curl http://localhost:18789` 返回 connection refused
- `ps aux | grep openclaw` 显示 gateway 进程在跑但 0% CPU
- 日志 `~/.openclaw/logs/gateway.log` 最后一条是 `"starting channels and sidecars..."` 然后停住
- 日志不写，但 `kill -USR2 <pid>` 可强制刷新日志缓冲区

### 根因（版本 2026.4.24 及之前）
OpenClaw gateway 启动时每次都重新安装所有插件的 bundled runtime deps（npm 包），共 66 个包，在 `~/.openclaw/plugin-runtime-deps/` 目录下。
- 正常需 8-12 分钟
- npm ENOTEMPTY 锁冲突时可达 30 分钟+
- 日志停在 "starting..." 是因为在等插件安装完成，并非真卡住
- 日志文件 600MB+ 会导致日志轮转出问题

### 根因（版本 2026.4.25 — 2026.4.26）
**browser 插件的 CDP 初始化会卡死 gateway**（0% CPU，端口监听但不响应 HTTP）。
- 进程 PID 存在，端口 18789 监听，但所有 HTTP 请求超时
- 日志不写入（或写入极慢）
- Dev 模式（`OPENCLAW_PROFILE=dev`）因 `skipBootstrap: true` 能起来

**修复**：在 `openclaw.json` 中关闭 browser 插件：
```python
import json
with open('/Users/mac/.openclaw/openclaw.json') as f:
    d = json.load(f)
d['plugins']['entries']['browser'] = {'enabled': False}
with open('/Users/mac/.openclaw/openclaw.json', 'w') as f:
    json.dump(d, f, indent=2)
```

### 根因（版本 2026.4.27+）— 升级后完全无法启动
**升级到 2026.4.27 后 gateway 卡在 "staging bundled runtime deps" 完全无响应。**

1. **plugin-runtime-deps 安装不完整**：同时存在 4 个版本目录（2026.4.24, 2026.4.25, 2026.4.27, unknown），版本之间文件互相引用导致缺失文件（如 `chutes-oauth-CUO6Kvd3.js`）
2. **npm 网络不通**：gateway 尝试安装 8 个缺失包（@agentclientprotocol/claude-agent-acp@0.31.0, @grammyjs/*, @homebridge/ciao@^1.3.6, acpx@0.6.1, chokidar@^5.0.0, grammy@^1.42.0 等），npm registry 访问超时后一直重试
3. **所有 openclaw CLI 命令全部超时**：`doctor`, `gateway stop`, `dashboard` 等都卡死
4. **launchctl 重启后 gateway 进程存在但不监听端口**：旧进程没真正被杀掉（launchd 管理混乱）

**Gateway token 位置**：`openclaw.json` 里存的是 `"***"`（生成后存在 `~/.openclaw/identity/device-auth.json` 的 `tokens.operator.token` 字段），该 token 是 device-to-device 认证，不用于 HTTP API。

## OpenClaw Gateway 运维

### 症状（版本 2026.4.24 及之前）
- `openclaw status` / `openclaw update check` 等命令全部超时（30s+）
- `curl http://localhost:18789` 返回 connection refused
- `ps aux | grep openclaw` 显示 gateway 进程在跑但 0% CPU
- 日志 `~/.openclaw/logs/gateway.log` 最后一条是 `"starting channels and sidecars..."` 然后停住
- 日志不写，但 `kill -USR2 <pid>` 可强制刷新日志缓冲区

### 根因（版本 2026.4.24 及之前）
OpenClaw gateway 启动时每次都重新安装所有插件的 bundled runtime deps（npm 包），共 66 个包，在 `~/.openclaw/plugin-runtime-deps/` 目录下。
- 正常需 8-12 分钟
- npm ENOTEMPTY 锁冲突时可达 30 分钟+
- 日志停在 "starting..." 是因为在等插件安装完成，并非真卡住
- 日志文件 600MB+ 会导致日志轮转出问题

### 根因（版本 2026.4.25 — 2026.4.26）
**browser 插件的 CDP 初始化会卡死 gateway**（0% CPU，端口监听但不响应 HTTP）。
- 进程 PID 存在，端口 18789 监听，但所有 HTTP 请求超时
- 日志不写入（或写入极慢）
- Dev 模式（`OPENCLAW_PROFILE=dev`）因 `skipBootstrap: true` 能起来

**修复**：在 `openclaw.json` 中关闭 browser 插件：
```python
import json
with open('/Users/mac/.openclaw/openclaw.json') as f:
    d = json.load(f)
d['plugins']['entries']['browser'] = {'enabled': False}
with open('/Users/mac/.openclaw/openclaw.json', 'w') as f:
    json.dump(d, f, indent=2)
```

### 根因（版本 2026.4.27+）— 升级后完全无法启动
**升级到 2026.4.27 后 gateway 卡在 "staging bundled runtime deps" 完全无响应。**

1. **plugin-runtime-deps 安装不完整**：同时存在 4 个版本目录（2026.4.24, 2026.4.25, 2026.4.27, unknown），版本之间文件互相引用导致缺失文件（如 `chutes-oauth-CUO6Kvd3.js`）
2. **npm 网络不通**：gateway 尝试安装 8 个缺失包（@agentclientprotocol/claude-agent-acp@0.31.0, @grammyjs/*, @homebridge/ciao@^1.3.6, acpx@0.6.1, chokidar@^5.0.0, grammy@^1.42.0 等），npm registry 访问超时后一直重试
3. **所有 openclaw CLI 命令全部超时**：`doctor`, `gateway stop`, `dashboard` 等都卡死
4. **launchctl 重启后 gateway 进程存在但不监听端口**：旧进程没真正被杀掉（launchd 管理混乱）

**Gateway token 位置**：`openclaw.json` 里存的是 `"***"`（生成后存在 `~/.openclaw/identity/device-auth.json` 的 `tokens.operator.token` 字段），该 token 是 device-to-device 认证，不用于 HTTP API。

## 执行模式（在 execute_code 或 terminal 中）

推荐在 `execute_code` 中用 Python subprocess 调用：

```python
import subprocess, json

def mmx_search(query, max_results=10):
    cmd = f'mmx search query --q "{query}" --output json'
    r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
    data = json.loads(r.stdout)
    results = []
    for item in data.get('organic', [])[:max_results]:
        results.append({
            'title': item.get('title', ''),
            'link': item.get('link', ''),
            'snippet': item.get('snippet', '').replace('</p><p>', ' ').replace('<p>', '').replace('</p>', ''),
            'date': item.get('date', '')
        })
    return results
```

## 搜索多条（多关键词合并去重）

```python
def search_briefing(keywords, max_results=5, max_total=12):
    all_results = {}
    for kw in keywords:
        try:
            results = mmx_search(kw, max_results=max_results)  # 注意参数名是 max_results，不是 max_per_kw
            for r in results:
                # 用 title 做去重 key
                key = r['title'][:30]
                if key not in all_results:
                    all_results[key] = r
        except Exception as e:
            print(f"搜索 '{kw}' 失败: {e}")
    
    items = list(all_results.values())
    items.sort(key=lambda x: x.get('date', ''), reverse=True)
    return items[:max_total]
```

## 飞书 Post 消息发送（踩坑总结）

飞书 post 消息 content 格式极易报 230001 错误（invalid message content），正确格式：

```python
import requests, json

# 正确的 content 格式：数组的数组 [[{...}], [{...}], ...]
# 每个内层数组 = 一个段落（paragraph），外层数组包含所有段落
content = [
    [{"tag": "text", "text": "第一行内容"}],
    [{"tag": "text", "text": "第二行内容"}],
    [{"tag": "text", "text": "第三行内容"}],
]

post_content = {
    "zh_cn": {
        "title": "标题",
        "content": content  # 直接放数组对象，不要再 json.dumps
    }
}

payload = {
    "receive_id": chat_id,
    "msg_type": "post",
    "content": json.dumps(post_content, ensure_ascii=False)
}

r = requests.post(
    "https://open.feishu.cn/open-apis/im/v1/messages?receive_id_type=chat_id",
    headers={
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json; charset=utf-8"
    },
    data=json.dumps(payload, ensure_ascii=False).encode("utf-8"),
    verify=False
)
```

关键点：
- content 是 `[[{...}]]`（数组的数组），不是 `[{...}]`（扁平数组）——错误格式会导致 230001
- content 字段本身是对象（直接放数组），不是字符串；外层 payload.content 才是 json.dumps 后的字符串
- 报错 230001 通常是 content 格式不对，不是 JSON 字符串或 token 问题
- 先用 text 类型消息测试 chat_id 和 token 是否正确
- 用 `requests` 而非 `urllib.request`——Python 3.9 urllib 对飞书文件上传和部分 POST 会持续返回 500，改用 requests 解决

## 新闻监控场景：提高搜索新鲜度的技巧

当搜索近期新闻时，以下策略可显著提升结果新鲜度：

1. **中文日期约束**：在 query 中加入具体日期的汉字形式，如 `"2026年5月9日 纳米"`，比 `"nanotech May 9 2026"` 返回结果更新
2. **中文实体名优先**：搜索中国科技新闻时，用中文专有名词直接搜索（如 `"汉原2号"`、`"本源悟空"`），英文名往往无结果
3. **多轮渐进式搜索**：先搜索宽泛关键词（如 `原子级制造 2026年5月`），再根据发现的实体名做二次精准搜索
4. **snippet 清洗**：返回的 snippet 含 HTML 标签，用 `re.sub(r'<[^>]+>', '', text)` 彻底去除，不要只用简单的字符串 replace
5. **搜索结果为空时的处理**：尝试去掉具体日期，改用 `"2026年5月 纳米制造 突破"` 等月级别约束，避免日期过细导致漏筛

def mmx_search_news(query, max_results=8):

                'date': item.get('date', '')

```python
import re

def md_to_html(md_text):
    """不含 emoji 过滤（Python 3.14 regex bug 会损坏 CJK 字符）"""
    html = md_text
    html = re.sub(r'^---\n[\s\S]+?\n---\n', '', html)
    html = re.sub(r'^# (.+)$', '<h1>\\1</h1>', html, flags=re.MULTILINE)
    html = re.sub(r'^## (.+)$', '<h2>\\1</h2>', html, flags=re.MULTILINE)
    html = re.sub(r'^### (.+)$', '<h3>\\1</h3>', html, flags=re.MULTILINE)
    html = re.sub(r'^#### (.+)$', '<h4>\\1</h4>', html, flags=re.MULTILINE)
    html = re.sub(r'^---$', '<hr>', html, flags=re.MULTILINE)
    html = re.sub(r'```(?:\w*)\n([\s\S]*?)\n```', '<pre><code>\\1</code></pre>', html)
    html = re.sub(r'`([^`]+)`', '<code>\\1</code>', html)
    html = re.sub(r'\*\*(.+?)\*\*', '<strong>\\1</strong>', html)
    html = re.sub(r'\*(.+?)\*', '<em>\\1</em>', html)
    html = re.sub(r'^> (.+)$', '<p><em>\\1</em></p>', html, flags=re.MULTILINE)
    # × 符号可能导致 500，标题级别在调用处处理
    lines = html.split('\n')
    in_table = False
    new_lines = []
    for line in lines:
        if '|' in line and line.strip().startswith('|'):
            cells = [c.strip() for c in line.strip().strip('|').split('|')]
            if not in_table:
                new_lines.append('<table border="1" cellpadding="8" cellspacing="0">')
                in_table = True
            is_header = all(re.match(r'[-:]+$', c.replace(' ', '')) for c in cells)
            tag = 'th' if is_header else 'td'
            new_lines.append('<tr>' + ''.join(
                f'<{tag}>{c}</{tag}>' for c in cells
                if not re.match(r'^[-:]+$', c.replace(' ', ''))
            ) + '</tr>')
        else:
            if in_table:
                new_lines.append('</table>')
                in_table = False
            new_lines.append(line)
    if in_table:
        new_lines.append('</table>')
    return '\n'.join(new_lines)

def extract_title(md):
    """提取标题并处理特殊字符"""
    m = re.search(r'^#\s+(.+)$', md, re.MULTILINE)
    title = m.group(1).strip() if m else '无标题'
    title = title.replace('×', 'x')   # × 会导致服务端 500
    title = title.replace('$', '美元')  # $ 也会导致服务端 500
    return title
```

| HTTP 500 稳定复现（标题含×） | `×`(U+00D7) 导致 JSON 解析失败 | 将 `×` 替换为 `x` 后重试 |
| HTML 变乱码（CJK被删） | Python 3.14 emoji 正则 bug 误删 CJK | 不要用 emoji 过滤函数，原文直传 |
| HTTP 500 err:ratelimit | 请求过于频繁，触发服务端限速 | 批量发布时每篇间隔 ≥10 秒，且必须单篇顺序发，不能并发 |
| HTTP 500 标题含特殊字符 | 标题含 `$`、`×` 等字符导致 JSON 解析失败 | 发布前替换：`.replace('$', '美元').replace('×', 'x')` |
